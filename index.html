Googleスプレッドシート連携設定

報告書データと現場写真をスプレッドシートへ自動記録

ステータス: 未連携

? 連携手順（初回のみ・約2分）

  1. 記録したいGoogleスプレッドシートを開き、上部メニュー「拡張機能」＞「Apps Script」を選択します。
  2. エディタ内の既存コードをすべて消し、下の「GASコードをコピー」したコードを貼り付けて保存します。
  3. 右上の「デプロイ」＞「新しいデプロイ」から種類で「ウェブアプリ」を選択し、アクセスできるユーザーを「全員」にしてデプロイします。
  4. 発行されたウェブアプリURLを上の入力欄に貼り付けて保存すると連携完了です。
※すでに連携済みでコードを更新する場合の重要手順：
Apps Scriptに新しいコードを貼り付けて保存しただけでは反映されません。必ずApps Script画面右上の「デプロイ」＞「デプロイを管理」＞ 右上の鉛筆（編集）アイコン ＞ バージョンで「新バージョン」を選択して「デプロイ」を押してください。
スプレッドシート用 Google Apps Script コード 
function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    // 複数端末からの同時送信でも行が欠落しないよう最大15秒待機
    lock.waitLock(15000);

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);

    var headers = [
      '実施日', '企業名', '店舗名', '担当者', '販売人数',
      '入店時間', '販売開始時間', '終了時間', '退店時間',
      '販売商品明細', '総販売準備数', '総販売数', '売上合計金額', 'メモ欄',
      '店舗外観写真URL', 'ポスター写真URL', '販売会写真URL', '送信日時'
    ];

    // テスト・デモ行の一括削除処理
    if (data.action === 'clearTestRows') {
      var lastRow = sheet.getLastRow();
      for (var r = lastRow; r >= 2; r--) {
        var rowValues = sheet.getRange(r, 1, 1, sheet.getLastColumn()).getValues()[0];
        var rowText = rowValues.join(' ');
        if (rowText.indexOf('【テスト送信】') !== -1 || rowText.indexOf('テスト担当者') !== -1 || rowText.indexOf('サンプル') !== -1) {
          sheet.deleteRow(r);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 旧形式（1列目が「送信日時」）が残っている場合は自動で最後の列へ整理
    if (sheet.getLastRow() >= 1) {
      var firstHeader = String(sheet.getRange(1, 1).getValue());
      if (firstHeader === '送信日時') {
        sheet.deleteColumn(1);
      }
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    } else {
      // 初回実行時にヘッダー行を自動作成（最後の列が「送信日時」）
      sheet.appendRow(headers);
    }

    // 現場写真をGoogleドライブへ保存してURL化
    var folderName = '販売会報告書_現場写真';
    var folders = DriveApp.getFoldersByName(folderName);
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);

    function saveBase64Image(dataUrl, label) {
      if (!dataUrl || dataUrl.indexOf('base64,') === -1) return 'なし';
      var parts = dataUrl.split(',');
      var contentType = parts[0].match(/:(.*?);/)[1];
      var bytes = Utilities.base64Decode(parts[1]);
      var fileName = (data.date || 'report') + '_' + (data.storeName || 'store') + '_' + label + '.jpg';
      var blob = Utilities.newBlob(bytes, contentType, fileName);
      var file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      return file.getUrl();
    }

    var photos = data.photos || {};
    var exteriorUrl = saveBase64Image(photos.exterior, '店舗外観');
    var posterUrl = saveBase64Image(photos.poster, 'ポスター');
    var eventUrl = saveBase64Image(photos.event, '販売会');

    // 最後の列（18列目）に「送信日時」を記録
    sheet.appendRow([
      data.date || '',
      data.companyName || '',
      data.storeName || '',
      data.staffName || '',
      data.headcount || 0,
      data.timeEnter || '',
      data.timeStart || '',
      data.timeEnd || '',
      data.timeExit || '',
      data.productSummary || '',
      data.totalPrepQty || 0,
      data.totalQty || 0,
      data.totalAmount || 0,
      data.memo || '',
      exteriorUrl,
      posterUrl,
      eventUrl,
      data.createdAt || new Date()
    ]);

    return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    if (lock.hasLock()) {
      lock.releaseLock();
    }
  }
}
※URL未設定でも「シート貼付コピー」「CSV保存」は利用可能です
